The .coverage.txt assets list inclusive hexadecimal character ranges from each
bundled font's characterSet, generated using pdfmake's fontkit 2.0.4 dependency.
They let PDF export fail explicitly for uncovered characters without bundling a
second font parser. Regenerate them whenever replacing either font:

node --input-type=module <<'JS'
import fs from 'node:fs';
import { create } from 'fontkit';
for (const name of ['NotoSansJP-Regular.otf', 'NotoEmoji.ttf']) {
  const font = create(fs.readFileSync(`src/assets/${name}`));
  const points = [...font.characterSet].sort((a, b) => a - b);
  const ranges = [];
  for (const point of points) {
    const last = ranges.at(-1);
    if (last && point === last[1] + 1) last[1] = point;
    else ranges.push([point, point]);
  }
  fs.writeFileSync(`src/assets/${name}.coverage.txt`, ranges.map(([start, end]) => `${start.toString(16)}-${end.toString(16)}`).join(',') + '\n');
}
